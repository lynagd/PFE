from rest_framework import serializers
from .models import Ordonnance

class ProduitPrescritSerializer(serializers.Serializer):
    nom = serializers.CharField(max_length=100)
    dosage = serializers.CharField(max_length=50)
    forme = serializers.CharField(max_length=50)
    quantite = serializers.IntegerField(default=1)
    instruction = serializers.CharField(max_length=255, allow_blank=True, required=False)

class OrdonnanceCreateSerializer(serializers.ModelSerializer):
    produits_prescrits = ProduitPrescritSerializer(many=True)

    class Meta:
        model = Ordonnance
        fields = [
            'id', 'medecin', 'patient', 'date_prescription', 'codebarre',
            'produits_prescrits',  'instruction_supplementaire', 'created_at'
        ]
        read_only_fields = ['id', 'date_prescription', 'created_at']

    def create(self, validated_data):
        produits_data = validated_data.pop('produits_prescrits', [])
        ordonnance = Ordonnance.objects.create(**validated_data)
        ordonnance.produits_prescrits = produits_data
        ordonnance.save()
        return ordonnance

    def update(self, instance, validated_data):
        produits_data = validated_data.pop('produits_prescrits', None)
        for attr, value in validated_data.items():
            setattr(instance, attr, value)
        if produits_data is not None:
            instance.produits_prescrits = produits_data
        instance.save()
        return instance
